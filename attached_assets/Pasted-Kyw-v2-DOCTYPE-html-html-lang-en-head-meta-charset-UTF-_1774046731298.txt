        const optsList = document.getElementById('opts-list');
        const factLabel = document.getElementById('fact-label');

        optsList.innerHTML = '';
        card.classList.remove('fact-card');

        if (item.type === 'fact') {
            card.classList.add('fact-card');
            factLabel.style.display = 'block';
            qText.innerText = item.text;
            
            // "Got it" button
            let btn = document.createElement('button');
            btn.className = 'opt-btn';
            btn.innerText = "I didn't know that! (Continue)";
            btn.style.textAlign = "center";
            btn.style.fontWeight = "bold";
            btn.onclick = () => {
                state.score++; // Free point
                nextStep();
            };
            optsList.appendChild(btn);

        } else {
            factLabel.style.display = 'none';
            qText.innerText = item.data.q;
            
            // Shuffle options
            let opts = [...item.data.opts].sort(() => Math.random() - 0.5);
            
            opts.forEach(opt => {
                let btn = document.createElement('button');
                btn.className = 'opt-btn';
                btn.innerText = opt;
                btn.onclick = () => checkAnswer(btn, opt, item.data.a);
                optsList.appendChild(btn);
            });
        }
    }

    function checkAnswer(btn, selected, correct) {
        // Disable all buttons
        const all = document.querySelectorAll('.opt-btn');
        all.forEach(b => b.disabled = true);

        if (selected === correct) {
            btn.classList.add('correct');
            state.score++;
        } else {
            btn.classList.add('wrong');
            // Highlight correct
            all.forEach(b => {
                if(b.innerText === correct) b.classList.add('correct');
            });
        }
        
        setTimeout(nextStep, 1000); // Wait 1s then move
    }

    function nextStep() {
        state.qIndex++;
        renderQuestion();
    }

    // --- 6. END LEVEL LOGIC ---
    function endLevel() {
        const modal = document.getElementById('result-modal');
        const title = document.getElementById('modal-title');
        const msg = document.getElementById('modal-msg');
        const icon = document.getElementById('modal-icon');
        const btn = document.getElementById('modal-action-btn');
        const homeBtn = document.getElementById('modal-home-btn');
        const factReveal = document.getElementById('fact-reveal');

        modal.classList.add('active');
        msg.innerText = `You scored ${state.score}/10`;
        factReveal.innerText = ""; // Clear old

        // Logic
        if (state.score < 5) {
            // FAIL
            icon.innerText = "📚";
            title.innerText = "Study Hard!";
            title.style.color = "var(--error)";
            btn.innerText = "Try Level Again";
            btn.onclick = () => {
                modal.classList.remove('active');
                loadLevel(); // Restart same level
            };
        } else {
            // PASS
            triggerConfetti();
            icon.innerText = "🎉";
            title.innerText = "Congratulations!";
            title.style.color = "var(--success)";
            
            // Final Fact logic if Level 3
            if (state.level === 3) {
                btn.innerText = "Replay Category";
                factReveal.innerText = "Theme Fact: Did you know? Learning about other cultures increases empathy and global awareness!";
                btn.onclick = () => {
                    modal.classList.remove('active');
                    startGame(state.category); // Restart Category
                };
            } else {
                btn.innerText = `Go to Level ${state.level + 1}`;
                btn.onclick = () => {
                    modal.classList.remove('active');
                    state.level++;
                    loadLevel();
                };
            }
        }

        homeBtn.onclick = () => {
            modal.classList.remove('active');
            navTo('screen-continents');
        };
    }

    function triggerConfetti() {
        for(let i=0; i<30; i++) {
            let c = document.createElement('div');
            c.className = 'confetti';
            c.style.left = Math.random()*100 + '%';
            c.style.backgroundColor = `hsl(${Math.random()*360}, 100%, 50%)`;
            c.style.animationDuration = (Math.random()*2 + 1) + 's';
            document.body.appendChild(c);
            setTimeout(() => c.remove(), 3000);
        }
    }

    function confirmExit() {
        if(confirm("Quit current game?")) navTo('screen-categories');
    }
